def send_syslog(message,level=6,facility=3,host='30.94.0.169',port=514):
    sock = socket.socket(socket.AF_INET,socket.SOCK_DGRAM)
    data = '<%d>%s'%(level+facility*8,message)
    sock.sendto(data,(host,port))
    sock.close()

##############################################################
#get all the members from object name or object-group name
##############################################################

def interpretName(objectname):
	name_content = {}
	for object in objectname:
		name = object.text.split()[-1]
		contentlist = []
		for content in object.all_children:
			contentlist.append(content.text)
		name_content[name] = contentlist
	return name_content

###############################################################
#The types of "object" and "object-group" look like as follows,
#
#object network XXX
#  host X.X.X.X
#  subnet X.X.X.X X.X.X.X
#
#object-group network XXX
#  network-object object XXX
#  network-object host X.X.X.X
#  network-object X.X.X.X X.X.X.X
#
#getObjectIP() is to get the real ip addresses according to  the 
#object name and object-group name
################################################################

def getObjectIP(device):
	t = time.strftime('%m%d',time.localtime(time.time()))
	
	path_object="/wls/networkdaily/networkdaily/cmdb/firewall"+"/"+t+"show-run-object"
	path_objectgroup="/wls/networkdaily/networkdaily/cmdb/firewall"+"/"+t+"show-run-object-group"
	
	fn_object = path_object + "/" + device + "show-run-object.txt"
	fn_objectgroup = path_objectgroup+"/" + device + "show-run-object-group.txt"
	
	parse_object = CiscoConfParse(fn_object)
	parse_objectgroup = CiscoConfParse(fn_objectgroup)

	hostname = parse_object.find_objects(r"^object\snetwork")
        groupname = parse_objectgroup.find_objects(r"^object-group\snetwork")
	
	host_ip = interpretName(hostname)
	group_host = interpretName(groupname)

	object_ip = {}

	selfip = ["30.66.239.241","30.66.239.242","30.65.239.241","30.65.239.242","30.98.239.241","30.98.239.242","30.97.239.241","30.97.239.242"]

	for host,ip in host_ip.items():
		iplist = []
		for i in ip:
			if "host" in i and not i[len("host")+2:] in selfip:
				iplist.append(i[len("host")+2:])
			elif "subnet" in i:
				iplist.append(i[len("subnet")+2:])
		object_ip[host] = iplist

	for group,host in group_host.items():
		hostlist = []
		for i in host:
			if "network-object object" in i:
				object = i[len("network-object object")+2:]
				if object_ip.get(object):
					hostlist.append(object_ip.get(object))
			elif "network-object host" in i:
				if i[len("network-object host")+2:] not in selfip:
					hostlist.append(i[len("network-object host")+2:])
			elif "network-object" in i:
				hostlist.append(i[len("network-object")+2:])
		object_ip[group] = hostlist

	object_ip["any"] = "any"
	return object_ip		

######################################################
#convert ip addresses in an 1D or 2D array to a string
######################################################

def arr2D_to_str(arr):
	if arr == None:
		return None
	if type(arr) == str:
		return arr
	str_ret = ""
	arr_temp =[]
	for a in arr:
		if type(a) == list:
			str_temp = ",".join(a)
		else:
			str_temp = a
		arr_temp.append(str_temp)
	str_ret = ",".join(arr_temp)
	return str_ret


def vipChange(address):
	if len(address.split(",")) > 1:
		return address
        seg = address.split(".")
        if int(seg[2]) > 200:
                seg[2] = int(seg[2]) - 4
        return seg[0]+"."+seg[1]+"."+str(seg[2])+"."+seg[3]

#get the status of big gateway in the given datacenter 
######################################################### 

def getStatus(dc, snat, inBX, inWGQ):
	if (dc == "B" and snat in inBX) or (dc == "W" and snat in inWGQ):
		return "online"
	elif (dc == "B" and snat in inWGQ) or (dc == "W" and snat in inBX):
		return "offline"
	else:
		return "not in use"

def getActiveRoute():
        t = time.strftime('%m%d',time.localtime(time.time()))
        path_B = '/wls/networkdaily/networkdaily/showrunning'+'/'+t+"showrunning"+"/"+"B-G07-PRT-SW-003-running.txt"
        path_W = '/wls/networkdaily/networkdaily/showrunning'+'/'+t+"showrunning"+"/"+"W-C05-PRT-SW-003-running.txt"
        f_B = open(path_B,"r")
        f_W = open(path_W,"r")
        parse_B = CiscoConfParse(f_B)
        parse_W = CiscoConfParse(f_W)
        obj_B = parse_B.find_objects(r"^ip\sroute.*Vlan3065")
        obj_W = parse_W.find_objects(r"^ip\sroute.*Vlan3065")

        route_B = []
        route_W = []

        for obj in obj_B:
                route_B.append(obj.text.split()[2].split("/")[0])

        for obj in obj_W:
                route_W.append(obj.text.split()[2].split("/")[0])
        f_B.close()
        f_W.close()

        return route_B, route_W

def main():
	devicename = ["B-INT-FW-001","B-PRT-FW-001","W-INT-FW-001","W-PRT-FW-001"]
        threads=[]
	for device in devicename:
        	t=threading.Thread(target=getNat,args=(device,))
        	threads.append(t)
        for t in threads:
                t.setDaemon(True)
                t.start()
        for t in threads:
                t.join()
        print "All threads are finished "
	

if __name__=='__main__':
	lock = threading.Lock()
	main()
