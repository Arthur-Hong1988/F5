import sys

    host = "30.95.0.34"

    viewzone = {"view_name":"external", "zone_name":zone_name}

    TXTRecord = {"domain_name":name, "text":value, "ttl":600}
    api = bigsuds.BIGIP(host,username="admin",password="Ad@1hblsqt")
    if action=="add":
        print("{}{}".format("add",TXTRecord)) 
        api.Management.ResourceRecord.add_txt([viewzone],[[TXTRecord]])
    elif action=="del":
        print("{}{}".format("del",TXTRecord))
        api.Management.ResourceRecord.delete_txt([viewzone],[[TXTRecord]])
    
    flag = False
    ret = api.Management.ResourceRecord.get_rrs([viewzone])
    #print ret
    for r in ret[0]:
        if name in r:
            flag = True
            break
    if flag == False and action== "del":

    elif flag == True and action=="add":
